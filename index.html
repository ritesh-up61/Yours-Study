function triggerStartTest() {

function startTest() {
const stream = document.getElementById('stream-select').value;
const mode = document.getElementById('mode-select').value;
const selectedClass = document.getElementById('class-select').value;
const selectedSub = document.getElementById('subject-select').value;
const selectedChap = document.getElementById('chapter-select').value;
const historyKey = "yourstudy_answered_" + currentUser;
let answeredIDs = JSON.parse(localStorage.getItem(historyKey) || "[]");
let pool = [];
if (stream === 'school') {
// STRICT FILTER: Match class AND subject AND chapter
pool = masterQuestions.filter(q =>
q.target === 'school' &&
q.class === selectedClass &&
q.subject === selectedSub &&
q.chapter === selectedChap
);
// Fallback within SAME subject only
if (pool.length === 0) {
pool = masterQuestions.filter(q =>
q.target === 'school' &&
q.class === selectedClass &&
q.subject === selectedSub
);
}
} else {
pool = masterQuestions.filter(q => q.target === 'competitive');
}
let freshQuestions = pool.filter(q => !answeredIDs.includes(q.id));
if (freshQuestions.length === 0) {
freshQuestions = pool;
answeredIDs = [];
localStorage.setItem(historyKey, JSON.stringify([]));
}
let limitVal = document.getElementById('limit-select').value;
let count = mode === 'timed' ? 15 : (limitVal ? parseInt(limitVal) : 10);
activeQuestions = freshQuestions.slice(0, count);
if (activeQuestions.length === 0) {
alert("Questions for this chapter are currently being indexed. Please select another
chapter.");
return;
}
currentIndex = 0;
userAnswers = {};
markedForReview.clear();
document.getElementById('screen-setup').classList.add('hidden');
document.getElementById('screen-test').classList.remove('hidden');
if (mode === 'timed') {
timeRemaining = 20 * 60;
document.getElementById('timer-display').classList.remove('hidden');
startTimer();
} else {
document.getElementById('timer-display').classList.add('hidden');
}
renderQuestion();
}
function startTimer() {
clearInterval(timerInterval);
timerInterval = setInterval(() => {
timeRemaining--;
let m = Math.floor(timeRemaining / 60);
let s = timeRemaining % 60;
document.getElementById('timer-display').innerText = `${m}:${s < 10 ? '0' : ''}${s}`;
if (timeRemaining <= 0) {
clearInterval(timerInterval);
triggerFinishTest();
}
}, 1000);
}
// 4. TEST RENDERING (Zero feedback during the test)
function renderQuestion() {
const q = activeQuestions[currentIndex];
const isMarked = markedForReview.has(q.id);
document.getElementById('question-tracker').innerText =
`Q ${currentIndex + 1}/${activeQuestions.length} ${isMarked ? ' ★ [Review]' : ''}`;
document.getElementById('question-text').innerText = q.question;
const optContainer = document.getElementById('options-container');
optContainer.innerHTML = '';
q.options.forEach((opt, idx) => {
const btn = document.createElement('button');
btn.className = `option-btn ${userAnswers[q.id] === idx ? 'selected' : ''}`;
btn.innerText = opt;
btn.onclick = () => {
userAnswers[q.id] = idx;
renderQuestion();
};
optContainer.appendChild(btn);
});
}
function toggleMarkForReview() {
const qId = activeQuestions[currentIndex].id;
if (markedForReview.has(qId)) markedForReview.delete(qId);
else markedForReview.add(qId);
renderQuestion();
}
function nextQuestion() {
if (currentIndex < activeQuestions.length - 1) {
currentIndex++;
renderQuestion();
} else {
triggerFinishTest();
}
}
function triggerFinishTest() {
clearInterval(timerInterval);
showLoader("Evaluating Test & Compiling Solutions...", () => {
finishTest();
});
}
// 5. POST-TEST EVALUATION
function finishTest() {
if (window.WTN && window.WTN.admob) {
window.WTN.admob.showInterstitial();
}
const historyKey = "yourstudy_answered_" + currentUser;
let answeredIDs = JSON.parse(localStorage.getItem(historyKey) || "[]");
activeQuestions.forEach(q => {
if (!answeredIDs.includes(q.id)) answeredIDs.push(q.id);
});
localStorage.setItem(historyKey, JSON.stringify(answeredIDs));
document.getElementById('screen-test').classList.add('hidden');
document.getElementById('screen-result').classList.remove('hidden');
let correctCount = 0;
const reviewList = document.getElementById('review-list');
reviewList.innerHTML = '';
activeQuestions.forEach((q, i) => {
const isCorrect = userAnswers[q.id] === q.correct;
if (isCorrect) correctCount++;
const item = document.createElement('div');
item.className = 'card';
item.innerHTML = `
<p><b>Q${i+1}: ${q.question}</b></p>
<p style="margin-top: 6px; font-size: 13px;">Your Choice:
<b>${q.options[userAnswers[q.id]] || "Not Attempted"}</b> | Correct: <b style="color:
#38bdf8;">${q.options[q.correct]}</b></p>
<div class="solution-box"><b>Shortcut Trick / Explanation:</b><br>${q.trickSolution}</div>
`;
reviewList.appendChild(item);
});
document.getElementById('score-text').innerText = `${correctCount} /
${activeQuestions.length}`;
document.getElementById('accuracy-text').innerText = `Accuracy:
${Math.round((correctCount / activeQuestions.length) * 100)}%`;
}
function resetToMenu() {
document.getElementById('screen-result').classList.add('hidden');
document.getElementById('screen-setup').classList.remove('hidden');
}
